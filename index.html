def init_db():
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS users (
            user_id INTEGER PRIMARY KEY,
            username TEXT,
            full_name TEXT,
            instagram TEXT,
            has_house INTEGER DEFAULT 0,
            house_type TEXT,
            car_name TEXT DEFAULT 'Joq'
        )
    """)
    try:
        cursor.execute("ALTER TABLE users ADD COLUMN car_name TEXT DEFAULT 'Joq'")
    except sqlite3.OperationalError:
        pass

    cursor.execute("""
        CREATE TABLE IF NOT EXISTS orders (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            order_type TEXT,
            details TEXT,
            amount INTEGER,
            screenshot_id TEXT,
            status TEXT DEFAULT 'pending',
            sent_otchet INTEGER DEFAULT 0,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)
    conn.commit()
    conn.close()

init_db()

def db_get_user(user_id):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("SELECT * FROM users WHERE user_id = ?", (user_id,))
    user = cursor.fetchone()
    conn.close()
    return user

def db_save_user(user_id, username, full_name, instagram):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("""
        INSERT INTO users (user_id, username, full_name, instagram) 
        VALUES (?, ?, ?, ?)
        ON CONFLICT(user_id) DO UPDATE SET instagram=excluded.instagram, username=excluded.username, full_name=excluded.full_name
    """, (user_id, username, full_name, instagram))
    conn.commit()
    conn.close()

def db_update_house_status(user_id, house_type):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("UPDATE users SET has_house = 1, house_type = ? WHERE user_id = ?", (house_type, user_id))
    conn.commit()
    conn.close()

def db_update_car_status(user_id, car_name):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("UPDATE users SET car_name = ? WHERE user_id = ?", (car_name, user_id))
    conn.commit()
    conn.close()

def db_add_order(user_id, order_type, details, amount, screenshot_id):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("""
        INSERT INTO orders (user_id, order_type, details, amount, screenshot_id)
        VALUES (?, ?, ?, ?, ?)
    """, (user_id, order_type, details, amount, screenshot_id))
    conn.commit()
    order_id = cursor.lastrowid
    conn.close()
    return order_id

def db_update_order_status(order_id, status):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("UPDATE orders SET status = ? WHERE id = ?", (status, order_id))
    cursor.execute("SELECT user_id, order_type, details FROM orders WHERE id = ?", (order_id,))
    res = cursor.fetchone()
    conn.close()
    return res

def db_get_unreported_orders():
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("""
        SELECT o.id, o.user_id, u.instagram, u.house_type, u.car_name, o.order_type, o.details, o.amount, o.created_at 
        FROM orders o 
        JOIN users u ON o.user_id = u.user_id 
        WHERE o.status = 'approved' AND o.sent_otchet = 0
    """)
    rows = cursor.fetchall()
    conn.close()
    return rows

def db_mark_orders_reported(order_ids):
    if not order_ids:
        return
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.executemany("UPDATE orders SET sent_otchet = 1 WHERE id = ?", [(oid,) for oid in order_ids])
    conn.commit()
    conn.close()
